package CaixaEletronico;

import java.util.Random;
import java.util.Scanner;

public class CaixaEletronico {

	public static void main(String[] args) {
		
		Scanner sc = new Scanner(System.in);
		Random rd = new Random();
		
		int opcao = 0;
		float saldo = 0;
		int senha = 1245;
		
		for(int x=0; x < 3; x++) {
			System.out.println("Digite a senha:");
			senha = sc.nextInt();
		if(senha == 1234) {
			
		
			do{
			
				System.out.println("Menu de opcoes:");
				System.out.println("Escolha uma das opcoes");
				System.out.println("----------------------");
				System.out.println("1.Consultar saldo");
				System.out.println("2.Depositar dinheiro");
				System.out.println("3.Sacar dinheiro");
				System.out.println("4.Sair");
				System.out.println("----------------------");
				
				opcao = sc.nextInt();
				
				
					
					float testeSaldo = 0;
					
					switch (opcao) {
					
						case 1:
							System.out.printf("Saldo: R$ %.2f%n", saldo);
						break;
						case 2:
							System.out.println("Digite o dinheiro a depositar:");
							testeSaldo = sc.nextFloat();
							if(testeSaldo <= 0) {
								System.out.println("Valor digitado invalido");
							}
							else {
								saldo = testeSaldo + saldo;
								System.out.printf("Valor depositado: R$ %.2f%n ", testeSaldo);
								System.out.printf("Saldo atual: R$ %.2f%n ", saldo);
							}
						break;
						case 3:
							System.out.println("Digite o valor a sacar:");
							float sacar = sc.nextFloat();
							
							if(sacar > saldo) {
								System.out.println("Valor indisponivel no saldo");
							}
							else if(sacar > 500){
								System.out.println("Valor limite diario");
							}	
							else {
								saldo = saldo - sacar;
								System.out.printf("Voce sacou: R$ %.2f%n ", sacar);
								System.out.printf("Saldo atual: R$ %.2f%n ", saldo);
						break;
							}
							
						default:
							System.out.println("Opção inválida! ");
					}
				
			
			}while(opcao != 4);
			
			System.out.println("Fim do progrma!!!");
		
			System.exit(0);
		}
		}
		
		System.out.println("Voce errou a senha 3 veses");

	}

}
